data Option A = None | Some of A

data Triple = A | B | C

let test1 x =
  match x with
  | Some (A | B) => "A or B"
  | Some C => "C"
  | None => "None"
  end

let test2 x =
  match x with
  | Some (Some (A | B | C)) => "nested triple"
  | Some None => "some none"
  | None => "none"
  end

let test3 x =
  match x with
  | Some (A | B) | None => "A, B, or None"
  | Some C => "C"
  end
